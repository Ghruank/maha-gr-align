# EXACT PIN, for two independent reasons. Both were hit; do not bump casually.
#
#  1. transformers 5.x moved `tokenization_utils.PreTrainedTokenizerBase`, which
#     IndicTransToolkit 1.1.1 imports — the toolkit fails to import at all.
#  2. IndicTrans2 ships its own modeling code (trust_remote_code) that still uses the
#     legacy tuple KV-cache API (`past_key_values[0][0].shape[2]`). transformers 4.57
#     hands it a Cache object instead, and generation dies with
#     `AttributeError: 'NoneType' object has no attribute 'shape'`.
#     4.53.3 still passes the legacy format. Verified translating both directions.
#
# If you must run a newer transformers, IndicTrans2 generation works with
# `use_cache=False` — same output, but measured 2.4x slower (4.8 s vs 2.0 s).
transformers==4.53.3

# Required by the IndicTrans2 tokenizer. Not pulled in automatically.
sentencepiece==0.2.2

sentence-transformers==5.6.1
accelerate==1.14.0
